const BLYPE = { viewer: localStorage.getItem("currentUser"), target: localStorage.getItem("profileTarget"), db: JSON.parse(localStorage.getItem("blypo_users") || "{}") }; function loadProfile() { const user = BLYPE.db[BLYPE.target]; if (!user) return; document.getElementById('p-username').innerText = BLYPE.target; // Load Friend Requests Database let requests = JSON.parse(localStorage.getItem("blypo_requests") || "[]"); const viewerData = BLYPE.db[BLYPE.viewer] || {}; const viewerFriends = viewerData.friends || []; const actionDiv = document.getElementById('social-actions'); // Check if a request is already sent const sentRequest = requests.find(r => r.from === BLYPE.viewer && r.to === BLYPE.target); const receivedRequest = requests.find(r => r.from === BLYPE.target && r.to === BLYPE.viewer); if (BLYPE.viewer === BLYPE.target) { actionDiv.innerHTML = ``; } else if (viewerFriends.includes(BLYPE.target)) { actionDiv.innerHTML = ` `; } else if (sentRequest) { actionDiv.innerHTML = ``; } else if (receivedRequest) { actionDiv.innerHTML = ``; } else { actionDiv.innerHTML = ` `; } // (Rest of your render logic for friends/badges/inventory remains the same) } function sendFriendRequest() { if (!BLYPE.viewer) return alert("Log in first!"); let requests = JSON.parse(localStorage.getItem("blypo_requests") || "[]"); requests.push({ from: BLYPE.viewer, to: BLYPE.target, status: "pending", id: Date.now() }); localStorage.setItem("blypo_requests", JSON.stringify(requests)); alert("Friend request sent to " + BLYPE.target + "!"); location.reload(); }
